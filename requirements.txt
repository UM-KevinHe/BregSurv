# Python runtime dependencies for the BregSurv agent UI.
# Pinned to versions verified locally on 2026-05-27 (anaconda Python 3.8)
# and intended to install cleanly on python:3.11-slim in Docker / HF Space.

# Gradio 5.x — we tried 4.44 first and hit a cascade of version-mismatch
# bugs (HfFolder removed, gradio_client bool-schema bug, starlette
# TemplateResponse API change). Gradio 5.x bundles current versions of
# its entire dep tree (starlette, fastapi, jinja2, huggingface_hub,
# gradio_client) so version pins are no longer load-bearing.
gradio>=5.0,<6.0
openai>=1.0.0
pandas>=1.3
matplotlib>=3.5
Pillow>=9.0

# Report rendering — PDF generation is wrapped in try/except in
# survbregdiv_agent/report.py so a missing weasyprint degrades to
# markdown-only download without crashing the UI. We still pin it here
# because the Docker image has pango/cairo and we want PDF on by default.
weasyprint==61.2
pydyf==0.10.0
markdown==3.7
